"use client";

import AppHeader from "@/layout/AppHeader";
import React, { useEffect } from "react";
import { usePathname, useRouter } from "@/i18n/navigation";
import { CURRENT_USER_STORAGE_KEY, DEFAULT_USERS, type ManagedUser } from "@/data/userData";

export default function AdminLayout({
  children,
}: {
  children: React.ReactNode;
}) {
  const pathname = usePathname();
  const router = useRouter();
  useEffect(() => {
    try {
      const raw = window.localStorage.getItem(CURRENT_USER_STORAGE_KEY);
      const user = raw ? JSON.parse(raw) as ManagedUser : DEFAULT_USERS[0];
      const routeToAccess: Record<string, string> = {
        "/": "dashboard", "/form-input-kecelakaan": "form-input-kecelakaan",
        "/rekap-kecelakaan": "rekap-kecelakaan", "/master-data": "master-data",
        "/manajemen-user": "manajemen-user", "/profile": "profile",
      };
      const required = routeToAccess[pathname];
      const allowed = user.accessPages !== undefined ? user.accessPages : Object.values(routeToAccess);
      if (required && !allowed.includes(required)) router.replace("/");
    } catch { /* Biarkan halaman tetap tampil jika data lokal belum tersedia. */ }
  }, [pathname, router]);
  return (
    <div className="min-h-screen w-full">
      <div className="w-full min-w-0">
        <AppHeader />
        <main className="w-full min-w-0 p-4 md:p-6">
          {children}
        </main>
      </div>
    </div>
  );
}

function getRecordDate(record: AccidentRecord) {
  if (record.tanggal) {
    return record.tanggal;
  }

  return record.createdAt?.slice(0, 10) || "";
}

function getRecordMonth(record: AccidentRecord) {
  const date = getRecordDate(record);
  const parsed = new Date(`${date}T00:00:00`);

  return Number.isNaN(parsed.getTime())
    ? -1
    : parsed.getMonth();
}

function getMonthYear(record: AccidentRecord) {
  const date = getRecordDate(record);
  const parsed = new Date(`${date}T00:00:00`);

  return Number.isNaN(parsed.getTime())
    ? "Tanggal tidak tersedia"
    : `${MONTHS[parsed.getMonth()]} ${parsed.getFullYear()}`;
}

function getRecordTimeValue(record: AccidentRecord) {
  if (!record.waktu) {
    return 0;
  }

  const parts = String(record.waktu)
    .split(":")
    .map(Number);

  const hour = parts[0] || 0;
  const minute = parts[1] || 0;
  const second = parts[2] || 0;

  return hour * 3600 + minute * 60 + second;
}

function formatDate(value: string) {
  if (!value) {
    return "-";
  }

  const date = new Date(
    `${value}T00:00:00`
  );

  return Number.isNaN(date.getTime())
    ? value
    : date.toLocaleDateString(
      "id-ID",
      {
        day: "2-digit",
        month: "long",
        year: "numeric",
      }
    );
}

function countBy<T>(
  items: T[],
  getKey: (item: T) => string
) {
  return items.reduce<
    Record<string, number>
  >((acc, item) => {
    const key =
      getKey(item) || "Tidak diisi";

    acc[key] =
      (acc[key] ?? 0) + 1;

    return acc;
  }, {});
}

function GenderPie({
  male,
  female,
  malePercentage,
  femalePercentage,
}: {
  male: number;
  female: number;
  malePercentage: number;
  femalePercentage: number;
}) {
  const [
    hoveredIndex,
    setHoveredIndex,
  ] = useState<number | null>(null);

  const total = male + female;

  if (!total) {
    return (
      <div className="flex flex-col items-center">
        <div className="relative flex w-full justify-center">
          <div className="relative">
            <svg
              width="180"
              height="180"
              viewBox="0 0 240 240"
              className="h-auto w-[180px] max-w-full"
            >
              <circle
                cx="120"
                cy="120"
                r="90"
                fill="none"
                stroke="#E5E7EB"
                strokeWidth="28"
              />
            </svg>
          </div>
        </div>
      </div>
    );
  }

  const maleAngle =
    (male / total) * 360;

  const femaleAngle =
    (female / total) * 360;

  const createArc = (
    startAngle: number,
    endAngle: number
  ) => {
    const centerX = 120;
    const centerY = 120;
    const radius = 90;

    const start =
      (Math.PI / 180) *
      (startAngle - 90);

    const end =
      (Math.PI / 180) *
      (endAngle - 90);

    const x1 =
      centerX +
      radius * Math.cos(start);

    const y1 =
      centerY +
      radius * Math.sin(start);

    const x2 =
      centerX +
      radius * Math.cos(end);

    const y2 =
      centerY +
      radius * Math.sin(end);

    const largeArcFlag =
      endAngle - startAngle > 180
        ? 1
        : 0;

    return [
      `M ${centerX} ${centerY}`,
      `L ${x1} ${y1}`,
      `A ${radius} ${radius} 0 ${largeArcFlag} 1 ${x2} ${y2}`,
      "Z",
    ].join(" ");
  };

  const malePath = createArc(
    0,
    maleAngle
  );

  const femalePath = createArc(
    maleAngle,
    maleAngle + femaleAngle
  );

  const maleIsHovered =
    hoveredIndex === 0;

  const femaleIsHovered =
    hoveredIndex === 1;

  return (
    <div className="flex flex-col items-center">
      <div className="relative flex w-full justify-center">
        <div className="relative">
          <svg
            width="180"
            height="180"
            viewBox="0 0 240 240"
            className="h-auto w-[180px] max-w-full"
          >
            <circle
              cx="120"
              cy="120"
              r="90"
              fill="#F3F4F6"
            />

            <g className="dashboard-pie-draw">
              <path
                d={malePath}
                fill="#16843F"
                opacity={
                  femaleIsHovered
                    ? 0.45
                    : 1
                }
                onMouseEnter={() =>
                  setHoveredIndex(0)
                }
                onMouseLeave={() =>
                  setHoveredIndex(null)
                }
                className="cursor-pointer transition-all duration-200"
                style={{
                  transformOrigin:
                    "120px 120px",
                  transform:
                    maleIsHovered
                      ? "scale(1.04)"
                      : "scale(1)",
                }}
              />

              <path
                d={femalePath}
                fill="#F28C45"
                opacity={
                  maleIsHovered
                    ? 0.45
                    : 1
                }
                onMouseEnter={() =>
                  setHoveredIndex(1)
                }
                onMouseLeave={() =>
                  setHoveredIndex(null)
                }
                className="cursor-pointer transition-all duration-200"
                style={{
                  transformOrigin:
                    "120px 120px",
                  transform:
                    femaleIsHovered
                      ? "scale(1.04)"
                      : "scale(1)",
                }}
              />

            </g>

            <circle
              cx="120"
              cy="120"
              r="62"
              fill="white"
            />

            <text
              x="120"
              y="114"
              textAnchor="middle"
              fontSize="13"
              fill="#6B7280"
            >
              Total
            </text>

            <text
              x="120"
              y="136"
              textAnchor="middle"
              fontSize="23"
              fontWeight="700"
              fill="#111827"
            >
              {total}
            </text>
          </svg>

          {hoveredIndex !== null && (
            <div className="pointer-events-none absolute left-1/2 top-1/2 z-20 -translate-x-1/2 -translate-y-1/2 whitespace-nowrap rounded-lg bg-white px-3 py-2 text-xs font-medium text-gray-800 shadow-lg">
              {hoveredIndex === 0
                ? `Male â€” ${male} (${malePercentage.toFixed(
                  1
                )}%)`
                : `Female â€” ${female} (${femalePercentage.toFixed(
                  1
                )}%)`}
            </div>
          )}
        </div>
      </div>

      <div className="mt-4 flex items-center justify-center gap-6">
        <div className="flex items-center gap-2">
          <span
            className="h-3 w-3 rounded-[3px]"
            style={{
              background: "#16843F",
            }}
          />

          <span className="text-xs text-gray-600 dark:text-gray-300">
            Male
          </span>
        </div>

        <div className="flex items-center gap-2">
          <span
            className="h-3 w-3 rounded-[3px]"
            style={{
              background: "#F28C45",
            }}
          />

          <span className="text-xs text-gray-600 dark:text-gray-300">
            Female
          </span>
        </div>
      </div>
    </div>
  );
}

function PieCard({
  title,
  subtitle,
  items,
}: {
  title: string;
  subtitle: string;
  items: [string, number][];
}) {
  const [
    hoveredIndex,
    setHoveredIndex,
  ] = useState<number | null>(null);

  const total = items.reduce(
    (sum, [, value]) => sum + value,
    0
  );

  if (!total) {
    return (
      <InsightCard
        title={title}
        subtitle={subtitle}
      >
        <div className="min-h-[260px]" />
      </InsightCard>
    );
  }
  const radius = 82;
  const center = 120;
  const circumference =
    2 * Math.PI * radius;

  let accumulated = 0;

  const segments = items.map(
    ([label, value], index) => {
      const percentage =
        value / total;

      const dash =
        percentage *
        circumference;

      const gap = 3;

      const segment = {
        label,
        value,
        percentage,
        dash,
        offset:
          -accumulated +
          circumference / 4,
        index,
      };

      accumulated += dash;

      return segment;
    }
  );

  return (
    <InsightCard
      title={title}
      subtitle={subtitle}
    >
      <div className="flex flex-col items-center gap-4 sm:flex-row sm:items-center sm:justify-center">
        <div className="relative shrink-0">
          <svg
            width="160"
            height="160"
            viewBox="0 0 240 240"
          >
            <circle
              cx="120"
              cy="120"
              r={radius}
              fill="none"
              stroke="#F3F4F6"
              strokeWidth="30"
            />

            <g className="dashboard-pie-draw">
              {segments.map(
                (segment) => {
                  const isHovered =
                    hoveredIndex ===
                    segment.index;

                  return (
                    <circle
                      key={segment.label}
                      cx="120"
                      cy="120"
                      r={radius}
                      fill="none"
                      stroke={
                        PIE_COLORS[
                        segment.index %
                        PIE_COLORS.length
                        ]
                      }

                        isHovered

                          : 30
                      }

                        0
                      )} ${circumference}`}
                      strokeDashoffset={
                        segment.offset
                      }
                      className="cursor-pointer transition-all duration-200"
                      onMouseEnter={() =>
                        setHoveredIndex(
                          segment.index
                        )
                      }
                      onMouseLeave={() =>
                        setHoveredIndex(null)
                      }
                    />
                  );
                }
              )}
            </g>

            <circle
              cx="120"
              cy="120"
              r="60"
              fill="white"
            />

            <text
              x="120"
              y="112"
              textAnchor="middle"
              fontSize="12"
              fill="#6B7280"
            >
              Total
            </text>

            <text
              x="120"
              y="137"
              textAnchor="middle"
              fontSize="23"
              fontWeight="700"
              fill="#111827"
            >
              {total}
            </text>
          </svg>

          {hoveredIndex !== null && (
            <div className="pointer-events-none absolute left-1/2 top-1/2 z-20 -translate-x-1/2 -translate-y-1/2 whitespace-nowrap rounded-lg bg-white px-3 py-2 text-xs font-medium text-gray-800 shadow-lg">
              {segments[
                hoveredIndex
              ]?.label}{" "}
              â€”{" "}
              {segments[

function SubDepartmentMonthlyChart({
  records,
  selectedSubDep,
  subDepOptions,
  analysisMode,
}: {
  records: AccidentRecord[];
  selectedSubDep: string;
  subDepOptions: string[];
  analysisMode: "monthly" | "yearly";
}) {
  const [
    hoveredIndex,
    setHoveredIndex,
  ] = useState<number | null>(null);

  const [
    tooltipPosition,
    setTooltipPosition,
  ] = useState({
    x: 0,
    y: 0,
  });

  const [
    selectedInsightPeriod,
    setSelectedInsightPeriod,
  ] = useState<number | null>(null);

  const selectedSubDepartmentDetailRef =
    useRef<HTMLDivElement>(null);

  useEffect(() => {
    if (selectedInsightPeriod === null) {
      return;
    }

    selectedSubDepartmentDetailRef.current?.scrollIntoView({
      behavior: "smooth",
      block: "start",
    });
  }, [selectedInsightPeriod]);

  const availableYears = useMemo(() => {
    const dataYears = Array.from(
      new Set(
        records
          .map((record) =>
            getRecordDate(record).slice(0, 4)
          )
          .filter((year) => /^\d{4}$/.test(year))
          .map(Number)
      )
    );

    // Mode tahunan selalu menampilkan 2026-2030 terlebih dahulu.
    // Jika nanti ada data tahun setelah 2030, tahun tersebut otomatis ikut muncul.
    const firstYear = 2026;
    const lastYear = Math.max(
      2030,
      ...dataYears,
    );

    return Array.from(
      { length: lastYear - firstYear + 1 },
      (_, index) => firstYear + index
    );
  }, [records]);

  const periodLabels = useMemo(() => {
    if (analysisMode === "yearly") {
      return availableYears.map(String);
    }

    return MONTHS.map((month) =>
      month.slice(0, 3)
    );
  }, [analysisMode, availableYears]);

  const periodFullLabels = useMemo(() => {
    if (analysisMode === "yearly") {
      return availableYears.map(String);
    }

    return MONTHS;
  }, [analysisMode, availableYears]);

  const periodCounts = useMemo(() => {
    if (analysisMode === "yearly") {
      // Mode tahunan menampilkan total seluruh kecelakaan dalam satu tahun.
      // Jadi 2026 = akumulasi Januari sampai Desember 2026, dan akan
      // otomatis bertambah ketika record kecelakaan baru masuk.
      // Filter SubDepartemen tetap digunakan pada mode bulanan.
      return availableYears.map((year) =>
        records.filter((record) => {
          const date = getRecordDate(record);

          if (!date) {
            return false;
          }

          return (
            Number(date.slice(0, 4)) === year &&
            (
              selectedSubDep === "all" ||
              String(record.worker?.subDep || "").trim() ===
              selectedSubDep
            )
          );
        }).length
      );
    }

    return MONTHS.map((_, monthIndex) => {
      return records.filter((record) => {
        const date = getRecordDate(record);

        if (!date) {
          return false;
        }

        if (
          getRecordMonth(record) !== monthIndex
        ) {
          return false;
        }

        if (
          selectedSubDep !== "all" &&
          String(
            record.worker?.subDep || ""
          ).trim() !== selectedSubDep
        ) {
          return false;
        }

        return true;
      }).length;
    });
  }, [
    records,
    selectedSubDep,
    analysisMode,
    availableYears,
  ]);

  const topSubDepartments = useMemo(() => {
    if (selectedInsightPeriod === null) {
      return [];
    }

    const selectedPeriodValue =
      analysisMode === "yearly"
        ? availableYears[selectedInsightPeriod]
        : selectedInsightPeriod;

    const counts = new Map<string, number>();

    subDepOptions.forEach((subDep) => {
      counts.set(subDep, 0);
    });

    records.forEach((record) => {
      const date = getRecordDate(record);

      if (!date) {
        return;
      }

      if (analysisMode === "yearly") {
        if (
          Number(date.slice(0, 4)) !==
          selectedPeriodValue
        ) {
          return;
        }
      } else if (
        getRecordMonth(record) !==
        selectedPeriodValue
      ) {
        return;
      }

      const subDep = String(
        record.worker?.subDep || ""
      ).trim();

      if (!subDep) {
        return;
      }

      counts.set(
        subDep,
        (counts.get(subDep) ?? 0) + 1
      );
    });

    return Array.from(counts.entries())
      .sort((a, b) => {
        if (b[1] !== a[1]) {
          return b[1] - a[1];
        }

        return a[0].localeCompare(b[0]);
      })
      .slice(0, 10);
  }, [
    records,
    selectedInsightPeriod,
    analysisMode,
    availableYears,
    subDepOptions,
  ]);

  const selectedSubDepPeriodCount = useMemo(() => {
    if (
      selectedInsightPeriod === null ||
      selectedSubDep === "all"
    ) {
      return 0;
    }

    if (analysisMode === "yearly") {
      const year =
        availableYears[selectedInsightPeriod];

      return records.filter((record) => {
        const date = getRecordDate(record);

        return (
          Number(date.slice(0, 4)) === year &&
          String(record.worker?.subDep || "").trim() ===
          selectedSubDep
        );
      }).length;
    }

    return records.filter((record) => {
      if (
        getRecordMonth(record) !==
        selectedInsightPeriod
      ) {
        return false;
      }

      return (
        String(record.worker?.subDep || "").trim() ===
        selectedSubDep
      );
    }).length;
  }, [
    records,
    selectedInsightPeriod,
    selectedSubDep,
    analysisMode,
    availableYears,
  ]);

  const selectedSubDepMonthlyRecap = useMemo(() => {
    if (selectedSubDep === "all") {
      return [];
    }

    return MONTHS.map((monthName, monthIndex) => {
      const count = records.filter((record) => {
        if (getRecordMonth(record) !== monthIndex) {
          return false;
        }

        return (
          String(record.worker?.subDep || "").trim() ===
          selectedSubDep
        );
      }).length;

      return {
        month: monthIndex,
        label: monthName,
        count,
      };
    });
  }, [records, selectedSubDep]);

  const selectedSubDepYearTotal = useMemo(() => {
    if (
      selectedSubDep === "all" ||
      selectedInsightPeriod === null ||
      analysisMode !== "yearly"
    ) {
      return 0;
    }

    const year =
      availableYears[selectedInsightPeriod];

    return records.filter((record) => {
      const date = getRecordDate(record);

      return (
        Number(date.slice(0, 4)) === year &&
        String(record.worker?.subDep || "").trim() ===
        selectedSubDep
      );
    }).length;
  }, [
    records,
    selectedSubDep,
    selectedInsightPeriod,
    analysisMode,
    availableYears,
  ]);

  const selectedYearHasData = useMemo(() => {
    if (
      selectedInsightPeriod === null ||
      analysisMode !== "yearly"
    ) {
      return false;
    }

    const year =
      availableYears[selectedInsightPeriod];

    return records.some((record) => {
      const date = getRecordDate(record);

      return (
        Number(date.slice(0, 4)) === year &&
        (
          selectedSubDep === "all" ||
          String(record.worker?.subDep || "").trim() ===
          selectedSubDep
        )
      );
    });
  }, [
    records,
    selectedSubDep,
    selectedInsightPeriod,
    analysisMode,
    availableYears,
  ]);

  const currentDate = new Date();
  const currentYear = currentDate.getFullYear();
  const currentMonth = currentDate.getMonth();

  const selectedMonthlyPeriodHasData = useMemo(() => {
    if (
      selectedInsightPeriod === null ||
      analysisMode !== "monthly"
    ) {
      return false;
    }

    return records.some((record) => {
      const date = getRecordDate(record);

      return (
        date &&
        getRecordMonth(record) === selectedInsightPeriod &&
        (
          selectedSubDep === "all" ||
          String(record.worker?.subDep || "").trim() ===
          selectedSubDep
        )
      );
    });
  }, [
    records,
    selectedSubDep,
    selectedInsightPeriod,
    analysisMode,
  ]);

  const selectedMonthlyPeriodIsFuture =
    analysisMode === "monthly" &&
    selectedInsightPeriod !== null &&
    selectedInsightPeriod > currentMonth;

  useEffect(() => {
    setSelectedInsightPeriod(null);
  }, [analysisMode, selectedSubDep]);

  /*
   * Animasi perubahan data.
   * Saat SubDepartemen diganti, nilai lama bergerak
   * secara halus menuju nilai baru seperti Campaign Performance.
   */
  const [
    animatedCounts,
    setAnimatedCounts,
  ] = useState<number[]>(periodCounts);

  const previousCountsRef =
    useRef<number[]>(periodCounts);

  const hasRenderedModeRef =
    useRef<Record<string, boolean>>({});

  const [lineRevealProgress, setLineRevealProgress] =
    useState(1);

  const lastModeRef = useRef(analysisMode);

  useEffect(() => {
    const modeChanged =
      lastModeRef.current !== analysisMode;

    const shouldDrawFromSide =
      !hasRenderedModeRef.current[analysisMode] ||
      modeChanged;

    lastModeRef.current = analysisMode;

    if (shouldDrawFromSide) {
      hasRenderedModeRef.current[analysisMode] = true;
      setLineRevealProgress(0);

      let revealFrame = 0;
      const revealStart = performance.now();
      const revealDuration = 3200;

      const reveal = (currentTime: number) => {
        const progress = Math.min(
          (currentTime - revealStart) / revealDuration,
          1
        );

        const eased =
          1 - Math.pow(1 - progress, 3);

        setLineRevealProgress(eased);

        if (progress < 1) {
          revealFrame = requestAnimationFrame(reveal);
        } else {
          setLineRevealProgress(1);
        }
      };

      revealFrame = requestAnimationFrame(reveal);

      return () => {
        cancelAnimationFrame(revealFrame);
      };
    }

    setLineRevealProgress(1);
  }, [analysisMode]);

  useEffect(() => {
    const from =
      previousCountsRef.current.length ===
        periodCounts.length
        ? previousCountsRef.current
        : periodCounts;

    const to = periodCounts;

    let animationFrame = 0;
    const startTime = performance.now();

    const duration = 3800;

    const easeInOutQuint = (value: number) => {
      return value < 0.5
        ? 16 * value * value * value * value * value
        : 1 - Math.pow(-2 * value + 2, 5) / 2;
    };

    const animate = (currentTime: number) => {
      const elapsed = currentTime - startTime;

      const progress = Math.min(
        elapsed / duration,
        1
      );

      const eased = easeInOutQuint(progress);

      const nextValues = to.map((target, index) => {
        const start = Number(from[index] ?? 0);

        return start + (target - start) * eased;
      });

      setAnimatedCounts(nextValues);

      if (progress < 1) {
        animationFrame = requestAnimationFrame(animate);
      } else {
        previousCountsRef.current = [...to];
        setAnimatedCounts([...to]);
      }
    };

    animationFrame = requestAnimationFrame(animate);

    return () => {
      cancelAnimationFrame(animationFrame);
    };
  }, [periodCounts]);

  const maxValue = Math.max(
    ...periodCounts,
    1
  );

  const animatedMaxValue = maxValue;
  const width = 1100;
  const height = 360;

  const paddingLeft = 52;
  const paddingRight = 28;
  const paddingTop = 24;
  const paddingBottom = 58;

  const chartWidth =
    width -
    paddingLeft -
    paddingRight;

  const chartHeight =
    height -
    paddingTop -
    paddingBottom;

  /*
   * Gunakan nilai animatedCounts untuk posisi garis.
   * Jadi path benar-benar bergerak dari posisi lama
   * menuju posisi baru.
   */
  const points = animatedCounts.map(
    (value, index) => {
      const x =
        paddingLeft +
        (index /
          Math.max(periodLabels.length - 1, 1)) *
        chartWidth;

      const y =
        paddingTop +
        chartHeight -
        (value / animatedMaxValue) *
        chartHeight;

      return {
        x,
        y,
        value,
      };
    }
  );

  const createSmoothPath = () => {
    if (!points.length) {
      return "";
    }

    if (points.length === 1) {
      return `M ${points[0].x} ${points[0].y}`;
    }

    let path =
      `M ${points[0].x} ${points[0].y}`;

    for (
      let index = 0;
      index < points.length - 1;
      index++
    ) {
      const current =
        points[index];

      const next =
        points[index + 1];

      const controlX =
        (current.x + next.x) / 2;

      path +=
        ` C ${controlX} ${current.y},` +
        ` ${controlX} ${next.y},` +
        ` ${next.x} ${next.y}`;
    }

    return path;
  };

  const linePath =
    createSmoothPath();

  const areaPath =
    points.length
      ? `${linePath} L ${points[points.length - 1].x
      } ${paddingTop +
      chartHeight
      } L ${points[0].x} ${paddingTop +
      chartHeight
      } Z`
      : "";

  const hoveredItem =
    hoveredIndex !== null
      ? {
        label:
          periodFullLabels[hoveredIndex] ??
          "",
        value:
          Math.round(
            animatedCounts[
            hoveredIndex
            ] ?? 0
          ),
      }
      : null;

  const tooltipWidth = 190;
  const tooltipHeight = 72;

  let tooltipX =
    tooltipPosition.x + 18;

  let tooltipY =
    tooltipPosition.y - 88;

  if (
    tooltipX + tooltipWidth >
    width - paddingRight
  ) {
    tooltipX =
      tooltipPosition.x -
      tooltipWidth -
      18;
  }

  if (
    tooltipX < paddingLeft
  ) {
    tooltipX = paddingLeft;
  }

  if (
    tooltipY < paddingTop
  ) {
    tooltipY =
      tooltipPosition.y + 18;
  }

  if (
    tooltipY + tooltipHeight >
    paddingTop + chartHeight
  ) {
    tooltipY =
      paddingTop +
      chartHeight -
      tooltipHeight;
  }

  return (
    <div className="w-full min-w-0">
      {/* Batasi seluruh animasi dan tooltip di area grafik agar tidak
          bergerak melewati batas kartu atau mendekati judul insight. */}
      <div className="relative w-full min-w-0 overflow-hidden pb-3">
        <svg
          width={width}
          height={height}
          viewBox={`0 0 ${width} ${height}`}
          className="block h-auto w-full min-w-0 overflow-hidden"
          role="img"
          aria-label={`Grafik kecelakaan kerja per ${analysisMode === "yearly"
            ? "tahun"
            : "bulan"
            } berdasarkan SubDepartemen`}
        >
          <defs>
            <linearGradient
              id="subDepartmentAreaGradient"
              x1="0"
              y1="0"
              x2="0"
              y2="1"
            >
              <stop
                offset="0%"
                stopColor="#F28C28"
                stopOpacity="0.16"
              />

              <stop
                offset="100%"
                stopColor="#F28C28"
                stopOpacity="0.01"
              />
            </linearGradient>

            <filter
              id="subDepartmentTooltipShadow"
              x="-30%"
              y="-30%"
              width="160%"
              height="180%"
            >
              <feDropShadow
                dx="0"
                dy="4"
                stdDeviation="6"
                floodColor="#000000"
                floodOpacity="0.10"
              />
            </filter>
          </defs>

          {[0, 25, 50, 75, 100].map(
            (percent) => {
              const y =
                paddingTop +
                chartHeight -
                (percent / 100) *
                chartHeight;

              const gridValue =
                Math.round(
                  (percent / 100) *
                  maxValue
                );

              return (
                <g
                  key={`subdep-grid-${percent}`}
                >
                  <line
                    x1={paddingLeft}
                    x2={
                      paddingLeft +
                      chartWidth
                    }
                    y1={y}
                    y2={y}
                    stroke="#e5e7eb"
                    strokeWidth="1"
                    strokeDasharray="4 5"
                  />

                  <text
                    x={
                      paddingLeft - 12
                    }
                    y={y + 4}
                    textAnchor="end"
                    fontSize="11"
                    fill="#9ca3af"
                  >
                    {gridValue}
                  </text>
                </g>
              );
            }
          )}

          <line
            x1={paddingLeft}
            x2={paddingLeft}
            y1={paddingTop}
            y2={
              paddingTop +
              chartHeight
            }
            stroke="#d1d5db"
            strokeWidth="1"
          />

          <line
            x1={paddingLeft}
            x2={
              paddingLeft +
              chartWidth
            }
            y1={
              paddingTop +
              chartHeight
            }
            y2={
              paddingTop +
              chartHeight
            }
            stroke="#d1d5db"
            strokeWidth="1"
          />

          {areaPath && (
            <path
              d={areaPath}
              fill="url(#subDepartmentAreaGradient)"
              pointerEvents="none"
            />
          )}

          <path
            d={linePath}
            fill="none"
            stroke="#F28C28"
            strokeWidth="3"
            strokeLinecap="round"
            strokeLinejoin="round"
            pathLength="1"
            strokeDasharray="1"
            strokeDashoffset={1 - lineRevealProgress}
          />

          {hoveredIndex !== null && (
            <line
              x1={
                points[
                  hoveredIndex
                ].x
              }
              x2={
                points[
                  hoveredIndex
                ].x
              }
              y1={paddingTop}
              y2={
                paddingTop +
                chartHeight
              }
              stroke="#9ca3af"
              strokeWidth="1"
              strokeDasharray="5 5"
              pointerEvents="none"
            />
          )}

          {points.map(
            (point, index) => {
              const previousPoint =
                points[index - 1];

              const nextPoint =
                points[index + 1];

              const leftBoundary =
                previousPoint
                  ? (previousPoint.x +
                    point.x) /
                  2
                  : paddingLeft;

              const rightBoundary =
                nextPoint
                  ? (point.x +
                    nextPoint.x) /
                  2
                  : paddingLeft +
                  chartWidth;

              const isHovered =
                hoveredIndex === index;

              return (
                <g
                  key={`subdep-point-${analysisMode}-${index}`}
                  onMouseEnter={() =>
                    setHoveredIndex(
                      index
                    )
                  }
                  onMouseMove={(event) => {
                    const rect =
                      event.currentTarget
                        .ownerSVGElement
                        ?.getBoundingClientRect();

                    if (!rect) {
                      return;
                    }

                    const scaleX =
                      width /
                      rect.width;

                    const scaleY =
                      height /
                      rect.height;

                    const x =
                      (event.clientX -
                        rect.left) *
                      scaleX;

                    const y =
                      (event.clientY -
                        rect.top) *
                      scaleY;

                    setTooltipPosition({
                      x,
                      y,
                    });
                  }}
                  onMouseLeave={() =>
                    setHoveredIndex(null)
                  }
                  onClick={() => {
                    setSelectedInsightPeriod(
                      selectedInsightPeriod === index
                        ? null
                        : index
                    );
                  }}
                  className="cursor-pointer"
                >
                  <rect
                    x={leftBoundary}
                    y={paddingTop}
                    width={Math.max(
                      rightBoundary -
                      leftBoundary,
                      1
                    )}
                    height={
                      chartHeight
                    }
                    fill="transparent"
                  />

                  <text
                    x={point.x}
                    y={
                      paddingTop +
                      chartHeight +
                      28
                    }
                    textAnchor="middle"
                    fontSize="11"
                    fontWeight={
                      isHovered
                        ? "600"
                        : "400"
                    }
                    fill={
                      isHovered
                        ? "#16a34a"
                        : "#6b7280"
                    }
                  >
                    {periodLabels[index]}
                  </text>
                </g>
              );
            }
          )}

          {hoveredItem && (
            <g
              pointerEvents="none"
              style={{
                transform: `translate(${tooltipX}px, ${tooltipY}px)`,
              }}
            >
              <rect
                x="0"
                y="0"
                width={tooltipWidth}
                height={tooltipHeight}
                rx="10"
                fill="#ffffff"
                stroke="#eef0f2"
                strokeWidth="1"
                filter="url(#subDepartmentTooltipShadow)"
              />

              <text
                x="16"
                y="24"
                fontSize="12"
                fontWeight="600"
                fill="#111827"
              >
                {hoveredItem.label}
              </text>

              <circle
                cx="20"
                cy="48"
                r="4"
                fill="#16a34a"
              />

              <text
                x="32"
                y="52"
                fontSize="12"
                fill="#6b7280"
              >
                Kecelakaan
              </text>

              <text
                x="145"
                y="52"
                fontSize="13"
                fontWeight="700"
                fill="#111827"
              >
                {hoveredItem.value}
              </text>
            </g>
          )}
        </svg>
      </div>

      {selectedInsightPeriod !== null && (
        <div
          ref={selectedSubDepartmentDetailRef}
          className="mt-4 rounded-xl border border-gray-100 bg-gray-50/50 px-5 py-5 dark:border-gray-800 dark:bg-gray-900/20"
        >
          <div className="mb-5 flex flex-col gap-2 sm:flex-row sm:items-start sm:justify-between">
            <div>
              <h3 className="text-sm font-semibold text-gray-800 dark:text-white/90">
                {selectedSubDep === "all"
                  ? `Top 10 SubDepartemen dengan Kecelakaan â€” ${periodFullLabels[selectedInsightPeriod]
                  }`
                  : `Kecelakaan ${selectedSubDep} â€” ${periodFullLabels[selectedInsightPeriod]
                  }`}
              </h3>

              <p className="text-xs text-gray-500 dark:text-gray-400">
                {selectedSubDep === "all"
                  ? `Menampilkan 10 SubDepartemen dengan jumlah kecelakaan tertinggi pada ${analysisMode === "yearly"
                    ? "tahun"
                    : "bulan"
                  } yang dipilih.`
                  : `Menampilkan jumlah kecelakaan ${selectedSubDep} pada ${analysisMode === "yearly"
                    ? "tahun"
                    : "bulan"
                  } yang dipilih.`}
              </p>
            </div>

            <button
              type="button"
              onClick={() =>
                setSelectedInsightPeriod(null)
              }
              className="ggf-close-filter shrink-0 self-start"
            >
              Tutup
            </button>
          </div>

          {selectedSubDep !== "all" ? (
            analysisMode === "yearly" ? (
              <div className="space-y-5">
                <div className="rounded-xl border border-gray-100 bg-white px-5 py-4 dark:border-gray-800 dark:bg-gray-900/30">
                  <div className="flex items-center justify-between gap-4">
                    <div>
                      <div className="text-xs text-gray-500 dark:text-gray-400">
                        Total Rekap Kecelakaan
                      </div>
                      <div className="mt-1 text-2xl font-semibold text-gray-800 dark:text-white/90">
                        {selectedSubDepYearTotal}
                      </div>
                    </div>

                    <div className="text-right text-xs text-gray-500 dark:text-gray-400">
                      {selectedSubDep}
                      <br />
                      {periodFullLabels[selectedInsightPeriod]}
                    </div>
                  </div>
                </div>

                {selectedSubDepYearTotal === 0 && (
                  <div className="rounded-xl border border-dashed border-gray-200 px-4 py-4 text-center text-xs text-gray-500 dark:border-gray-700 dark:text-gray-400">
                    Tidak ada kecelakaan pada tahun{" "}
                    {periodFullLabels[selectedInsightPeriod]}{" "}
                    untuk {selectedSubDep}.
                  </div>
                )}
              </div>
            ) : (
              <div className="space-y-5">
                <div className="rounded-xl border border-gray-100 bg-white px-5 py-4 dark:border-gray-800 dark:bg-gray-900/30">
                  <div className="flex items-center justify-between gap-4">
                    <div>
                      <div className="text-xs text-gray-500 dark:text-gray-400">
                        Total Rekap Kecelakaan
                      </div>
                      <div className="mt-1 text-2xl font-semibold text-gray-800 dark:text-white/90">
                        {selectedSubDepMonthlyRecap.reduce(
                          (sum, item) => sum + item.count,
                          0

                      </div>
                    </div>

                    <div className="text-right text-xs text-gray-500 dark:text-gray-400">
                      {selectedSubDep}
                      <br />
                      seluruh bulan
                    </div>
                  </div>
                </div>

                <div className="space-y-3">
                  {selectedSubDepMonthlyRecap.map((item) => {

                      ...selectedSubDepMonthlyRecap.map(
                        (entry) => entry.count
                      ),
                      1
                    );

                    const widthPercentage =
                      (item.count / maxCount) * 100;

                    const isSelected =
                      item.month === selectedInsightPeriod;

                    return (
                      <div
                        key={item.month}
                        className="grid grid-cols-[28px_minmax(150px,1fr)_minmax(120px,2fr)_44px] items-center gap-3"
                      >
                        <span
                          className={`text-xs font-semibold ${isSelected
                            ? "text-emerald-700 dark:text-emerald-400"
                            : "text-gray-500 dark:text-gray-400"
                            }`}
                        >
                          {item.month + 1}
                        </span>

                        <span
                          className={`truncate text-xs font-medium ${isSelected
                            ? "text-emerald-700 dark:text-emerald-400"
                            : "text-gray-700 dark:text-gray-300"
                            }`}
                          title={item.label}
                        >
                          {item.label}
                        </span>

                        <div className="h-2.5 overflow-hidden rounded-full bg-gray-100 dark:bg-gray-700/40">
                          <div
                            className="h-full rounded-full transition-all duration-500"
                            style={{
                              width: `${widthPercentage}%`,
                              background:
                                item.count > 0
                                  ? "#B8D8A8"
                                  : "transparent",
                              boxShadow:
                                isSelected && item.count > 0
                                  ? "0 2px 7px rgba(120, 170, 100, 0.35)"
                                  : "none",
                              transform: isSelected
                                ? "scaleY(1.18)"
                                : "scaleY(1)",
                              transformOrigin: "center",
                            }}
                          />
                        </div>

                        <span
                          className={`text-right text-xs font-semibold ${isSelected
                            ? "text-emerald-700 dark:text-emerald-400"
                            : "text-gray-800 dark:text-white/90"
                            }`}
                        >
                          {item.count}
                        </span>
                      </div>
                    );
                  })}
                </div>

                {selectedSubDepPeriodCount === 0 && (
                  <div className="rounded-xl border border-dashed border-gray-200 px-4 py-4 text-center text-xs text-gray-500 dark:border-gray-700 dark:text-gray-400">
                    Tidak ada kecelakaan pada{" "}
                    {periodFullLabels[selectedInsightPeriod]}{" "}
                    untuk {selectedSubDep}.
                  </div>
                )}
              </div>
            )
          ) : (
            (analysisMode === "yearly" &&
              (
                (availableYears[selectedInsightPeriod] ??
                  currentYear) > currentYear ||
                !selectedYearHasData
              )) ||
            (analysisMode === "monthly" &&
              (
                selectedMonthlyPeriodIsFuture ||
                !selectedMonthlyPeriodHasData
              ))
          ) ? (
            <div className="rounded-xl border border-dashed border-gray-200 px-4 py-8 text-center dark:border-gray-700">
              <p className="text-sm font-semibold text-gray-700 dark:text-gray-200">
                Data belum tersedia
              </p>
              <p className="mt-1 text-xs text-gray-500 dark:text-gray-400">
                {analysisMode === "yearly"
                  ? availableYears[selectedInsightPeriod] > currentYear
                    ? `Tahun ${availableYears[selectedInsightPeriod]} belum berjalan.`
                    : `Belum ada data kecelakaan untuk tahun ${availableYears[selectedInsightPeriod]}.`
                  : selectedMonthlyPeriodIsFuture
                    ? `Bulan ${periodFullLabels[selectedInsightPeriod]} belum berjalan.`
                    : `Belum ada data kecelakaan untuk bulan ${periodFullLabels[selectedInsightPeriod]}.`}
              </p>
            </div>
          ) : topSubDepartments.length > 0 ? (
            <div className="space-y-3">
              {topSubDepartments.map(
                ([subDep, count], index) => {
                  const maxCount = Math.max(
                    topSubDepartments[0]?.[1] ?? 0,
                    1
                  );

                  const widthPercentage =
                    count > 0 ? (count / maxCount) * 100 : 0;

                  return (
                    <div
                      key={subDep}
                      className="grid grid-cols-[28px_minmax(150px,1fr)_minmax(120px,2fr)_44px] items-center gap-3"
                    >
                      <span className="text-xs font-semibold text-gray-500 dark:text-gray-400">
                        {index + 1}
                      </span>

                      <span
                        className="truncate text-xs font-medium text-gray-700 dark:text-gray-300"
                        title={subDep}
                      >
                        {subDep}
                      </span>

                      <div className="h-2.5 overflow-hidden rounded-full bg-gray-100 dark:bg-gray-700/40">
                        <div
                          className="h-full rounded-full transition-all duration-500"
                          style={{
                            width: `${widthPercentage}%`,
                            background:
                              count > 0
                                ? "#B8D8A8"
                                : "transparent",
                          }}
                        />
                      </div>

                      <span className="text-right text-xs font-semibold text-gray-800 dark:text-white/90">
                        {count}
                      </span>
                    </div>
                  );
                }
              )}
            </div>
          ) : (
            <div className="rounded-xl border border-dashed border-gray-200 px-4 py-6 text-center text-sm text-gray-500 dark:border-gray-700 dark:text-gray-400">
              Tidak ada data kecelakaan pada{" "}
              {analysisMode === "yearly"
                ? "tahun"
                : "bulan"}{" "}
              ini.
            </div>
          )}
        </div>
      )}
    </div>
  );
}

function exportAnnualReport(
  records: AccidentRecord[],
  year: number
) {
  const annualRecords = records
    .filter((record) => {
      const date = getRecordDate(record);
      return Number(date.slice(0, 4)) === year;
    })
    .sort((a, b) => {
      const dateCompare = getRecordDate(a).localeCompare(
        getRecordDate(b)
      );

      if (dateCompare !== 0) {
        return dateCompare;
      }

      return getRecordTimeValue(a) - getRecordTimeValue(b);
    });

  const escapeXml = (value: unknown) =>
    String(value ?? "")
      .replace(/&/g, "&amp;")
      .replace(/</g, "&lt;")
      .replace(/>/g, "&gt;")
      .replace(/"/g, "&quot;")
      .replace(/'/g, "&apos;");

  // Urutan kolom mengikuti template Form Data Kecelakaan Kerja.
  const fields = [
    "Tanggal",
    "Waktu",
    "KIT",
    "Nama Lengkap",
    "Jenis Kelamin",
    "Umur",
    "SubDep",
    "Div/SubDiv",
    "Status Pekerja",
    "Wilayah / Lokasi",
    "Grouping",
    "Kronologi Kejadian",
    "Akibat",
    "Status Rujuk",
    "Type Of Accident",
  ];

  // Lebar dibuat mengikuti proporsi template Form Data.
  const columnWidths = [
    105,
    80,
    95,
    180,
    105,
    55,
    155,
    150,
    95,
    150,
    120,
    260,
    120,
    90,
    125,
  ];

  const exportMonths = [
    "January",
    "February",
    "March",
    "April",
    "May",
    "June",
    "July",
    "August",
    "September",
    "October",
    "November",
    "December",
  ];

  const formatExportDate = (value: string) => {
    if (!value) {
      return "";
    }

    const parts = value.split("-");
    if (parts.length !== 3) {
      return formatDate(value);
    }

    const yearPart = parts[0];
    const monthPart = Number(parts[1]);
    const dayPart = parts[2];

    if (
      !Number.isInteger(monthPart) ||
      monthPart < 1 ||
      monthPart > 12
    ) {
      return formatDate(value);
    }

    return `${dayPart} ${exportMonths[monthPart - 1]} ${yearPart}`;
  };

  const getRecordRows = (monthRecords: AccidentRecord[]) =>
    monthRecords.map((record) => [
      formatExportDate(getRecordDate(record)),
      record.waktu || "",
      record.worker?.kit || "",
      record.worker?.namaLengkap || "",
      record.worker?.jenisKelamin || "",
      record.worker?.umur || "",
      record.worker?.subDep || "",
      record.worker?.divSubDiv || "",
      record.worker?.statusPekerja || "",
      record.lokasi || "",
      record.grouping || "",
      record.kronologi || "",
      record.akibat || "",
      record.statusRujuk || "",
      record.typeAccident || "",
    ]);

  const makeCell = (value: unknown, styleId = "Data") =>
    `<Cell ss:StyleID="${styleId}"><Data ss:Type="String">${escapeXml(
      value
    )}</Data></Cell>`;

  const makeNumberCell = (value: number, styleId = "Number") =>
    `<Cell ss:StyleID="${styleId}"><Data ss:Type="Number">${value}</Data></Cell>`;

  const makeRow = (values: unknown[], styleId = "Data") =>
    `<Row ss:AutoFitHeight="0" ss:Height="20">${values
      .map((value) => makeCell(value, styleId))
      .join("")}</Row>`;

  const makeWorksheet = (
    name: string,
    rows: string,
    options?: {
      summary?: boolean;
    }
  ) => {
    const isSummary = options?.summary === true;
    const widths = isSummary ? [120, 125] : columnWidths;

    return `
    <Worksheet ss:Name="${escapeXml(name)}">
      <Table ss:ExpandedColumnCount="${widths.length}" ss:ExpandedRowCount="0" x:FullColumns="1" x:FullRows="1">
        ${widths
        .map((width) => `<Column ss:Width="${width}" />`)
        .join("")}
        ${rows}
      </Table>
    </Worksheet>`;
  };

  const summaryRows = MONTHS.map((month, monthIndex) => {
    const count = annualRecords.filter(
      (record) => getRecordMonth(record) === monthIndex
    ).length;

    return `<Row ss:AutoFitHeight="0" ss:Height="20">
      ${makeCell(month, "Data")}
      ${makeNumberCell(count)}
    </Row>`;
  }).join("");

  const summaryWorksheet = makeWorksheet(
    "Ringkasan",
    `<Row ss:AutoFitHeight="0" ss:Height="20">
      <Cell ss:MergeAcross="1" ss:StyleID="SummaryTitle">
        <Data ss:Type="String">Rekap kecelakaan kerja</Data>
      </Cell>
    </Row>
    <Row ss:AutoFitHeight="0" ss:Height="20">

      ${makeCell("Jumlah Kecelakaan", "Header")}
    </Row>
    ${summaryRows}
    <Row ss:AutoFitHeight="0" ss:Height="20">
      ${makeCell("TOTAL KECELAKAAN", "Total")}
      ${makeNumberCell(annualRecords.length, "TotalNumber")}
    </Row>`,
    { summary: true }
  );

  const monthlyWorksheets = MONTHS.map((month, monthIndex) => {
    const monthRecords = annualRecords.filter(
      (record) => getRecordMonth(record) === monthIndex
    );

    // Template Form Data dimulai langsung dari header pada baris pertama.
    const headerRow = `<Row ss:AutoFitHeight="0" ss:Height="22">
      ${fields.map((field) => makeCell(field, "Header")).join("")}
    </Row>`;

    if (!monthRecords.length) {
      return makeWorksheet(
        month,
        `${headerRow}
        <Row ss:AutoFitHeight="0" ss:Height="20">
          <Cell ss:MergeAcross="14" ss:StyleID="EmptyMessage">
            <Data ss:Type="String">Data kecelakaan belum tersedia.</Data>
          </Cell>
        </Row>`
      );
    }

    const dataRows = getRecordRows(monthRecords)
      .map((row) => makeRow(row))
      .join("");

    return makeWorksheet(month, `${headerRow}${dataRows}`);
  }).join("");

  const xml = `<?xml version="1.0"?>
<?mso-application progid="Excel.Sheet"?>
<Workbook xmlns="urn:schemas-microsoft-com:office:spreadsheet"
  xmlns:o="urn:schemas-microsoft-com:office:office"
  xmlns:x="urn:schemas-microsoft-com:office:excel"
  xmlns:ss="urn:schemas-microsoft-com:office:spreadsheet"
  xmlns:html="http://www.w3.org/TR/REC-html40">
  <Styles>
    <Style ss:ID="Default" ss:Name="Normal">
      <Alignment ss:Vertical="Center" />
      <Font ss:FontName="Calibri" ss:Size="11" />
    </Style>

    <Style ss:ID="Header">
      <Alignment ss:Horizontal="Center" ss:Vertical="Center" />
      <Font ss:FontName="Calibri" ss:Size="11" ss:Bold="1" />
      <Interior ss:Color="#00B0F0" ss:Pattern="Solid" />
    </Style>

    <Style ss:ID="Data">
      <Alignment ss:Vertical="Center" />
      <Font ss:FontName="Calibri" ss:Size="11" />
    </Style>

    <Style ss:ID="Number">
      <Alignment ss:Horizontal="Center" ss:Vertical="Center" />
      <Font ss:FontName="Calibri" ss:Size="11" />
    </Style>

    <Style ss:ID="SummaryTitle">
      <Alignment ss:Horizontal="Center" ss:Vertical="Center" />
      <Font ss:FontName="Calibri" ss:Size="11" ss:Bold="1" />
      <Interior ss:Color="#00B0F0" ss:Pattern="Solid" />
    </Style>

    <Style ss:ID="Total">
      <Alignment ss:Vertical="Center" />
      <Font ss:FontName="Calibri" ss:Size="11" ss:Bold="1" />
    </Style>

    <Style ss:ID="TotalNumber">
      <Alignment ss:Horizontal="Center" ss:Vertical="Center" />
      <Font ss:FontName="Calibri" ss:Size="11" ss:Bold="1" />
    </Style>

    <Style ss:ID="EmptyMessage">
      <Alignment ss:Vertical="Center" />
      <Font ss:FontName="Calibri" ss:Size="11" />
    </Style>
  </Styles>
  ${summaryWorksheet}
  ${monthlyWorksheets}
</Workbook>`;

  const blob = new Blob([xml], {
    type: "application/vnd.ms-excel;charset=utf-8;",
  });

  const url = URL.createObjectURL(blob);
  const link = document.createElement("a");

  link.href = url;
  link.download = `Rekap-Kecelakaan-${year}.xls`;

  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);

  URL.revokeObjectURL(url);
}

import { SidebarProvider } from "@/context/SidebarContext";
import { ThemeProvider } from "@/context/ThemeContext";
import { isRtl } from "@/i18n/languages";
import { type Locale, routing } from "@/i18n/routing";
import "flatpickr/dist/flatpickr.css";
import { NextIntlClientProvider } from "next-intl";
import { setRequestLocale } from "next-intl/server";
import { Outfit } from "next/font/google";
import { notFound } from "next/navigation";
import type { Metadata } from "next";
import "simplebar-react/dist/simplebar.min.css";
import "swiper/css/bundle";
import "../globals.css";

const outfit = Outfit({
  subsets: ["latin"],
});

export const metadata: Metadata = {
  title: "Rekap Kecelakaan Kerja | Great Giant Foods",
  description: "Dashboard dan rekap data kecelakaan kerja Great Giant Foods.",
  icons: {
    icon: "/images/logo/ggf-logo.png",
    shortcut: "/images/logo/ggf-logo.png",
    apple: "/images/logo/ggf-logo.png",
  },
};

export function generateStaticParams() {
  return routing.locales.map((locale) => ({ locale }));
}

export default async function RootLayout({
  children,
  params,
}: Readonly<{
  children: React.ReactNode;
  params: Promise<{ locale: string }>;
}>) {
  const { locale } = await params;

  if (!routing.locales.includes(locale as Locale)) {
    notFound();
  }

  setRequestLocale(locale);

  return (
    <html lang={locale} dir={isRtl(locale as Locale) ? "rtl" : "ltr"}>
      <body className={`${outfit.className} dark:bg-gray-900`}>
        <NextIntlClientProvider>
          <ThemeProvider>
            <SidebarProvider>{children}</SidebarProvider>
          </ThemeProvider>
        </NextIntlClientProvider>
      </body>
    </html>
  );
}
