===== APP HEADER =====
"use client";

import UserDropdown from "@/components/header/UserDropdown";
import { Link, usePathname } from "@/i18n/navigation";
import { cn } from "@/utils";
import Image from "next/image";
import { useEffect, useState } from "react";
import { CURRENT_USER_STORAGE_KEY, DEFAULT_USERS, type ManagedUser } from "@/data/userData";

const AppHeader: React.FC = () => {
  const pathname = usePathname();
  const [allowedPages, setAllowedPages] = useState<string[] | null>(null);
  useEffect(() => {
    const refresh = () => {
      try {
        const raw = window.localStorage.getItem(CURRENT_USER_STORAGE_KEY);
        const user = raw ? JSON.parse(raw) as ManagedUser : DEFAULT_USERS[0];
        setAllowedPages(user.accessPages !== undefined ? user.accessPages : ["dashboard", "form-input-kecelakaan", "rekap-kecelakaan", "master-data", "manajemen-user", "profile"]);
      } catch { setAllowedPages(["dashboard", "form-input-kecelakaan", "rekap-kecelakaan", "master-data", "manajemen-user", "profile"]); }
    };
    refresh(); window.addEventListener("ggf-user-updated", refresh);
    return () => window.removeEventListener("ggf-user-updated", refresh);
  }, []);

  const items = [
    { href: "/", label: "Dashboard", access: "dashboard" },
    { href: "/form-input-kecelakaan", label: "Form Input Kecelakaan", access: "form-input-kecelakaan" },
    { href: "/rekap-kecelakaan", label: "Data Rekap Kecelakaan", access: "rekap-kecelakaan" },
    { href: "/master-data", label: "Master Data", access: "master-data" },
  ].filter((item) => !allowedPages || allowedPages.includes(item.access));
  const canManageUsers = !allowedPages || allowedPages.includes("manajemen-user");

  return (
    <header className="sticky top-0 z-99999 w-full border-b border-[#e0e8e2] bg-white">
      <div className="mx-auto flex min-h-[76px] w-full items-center gap-4 px-4 py-2 sm:px-6 xl:px-9">
        <Link href="/" className="flex min-w-0 shrink-0 items-center gap-4" aria-label="Great Giant Foods">
          <Image width={112} height={62} className="h-12 w-auto object-contain" src="/images/logo/ggf-logo.png" alt="Great Giant Foods" priority />
          <span className="hidden min-w-0 sm:block">
            <span className="block whitespace-nowrap text-[15px] font-bold leading-5 text-[#183b2b]">Dashboard Kecelakaan Kerja</span>
            <span className="block whitespace-nowrap text-[11px] font-medium leading-4 text-[#7d8981]">Sistem Informasi Great Giant Foods</span>
          </span>
        </Link>

        <nav aria-label="Navigasi utama" className="min-w-0 flex-1 overflow-x-auto no-scrollbar">
          <div className="flex min-w-max items-stretch justify-center gap-1">
            {items.map(({ href, label }) => {
              const active = pathname === href;
              return (
                <Link key={href} href={href} className={cn(
                  "flex min-h-[52px] items-center justify-center whitespace-nowrap rounded-t-xl border-b-2 px-4 text-[12px] font-semibold transition-colors duration-150",
                  active ? "border-[#07813e] bg-[#eef8f1] text-[#07813e]" : "border-transparent text-[#657169] hover:bg-[#f5f8f5] hover:text-[#07813e]"
                )}>{label}</Link>
              );
            })}
          </div>
        </nav>

        <div className="ml-auto flex shrink-0 items-center gap-2">
          <UserDropdown showLogoutButton />
          {canManageUsers && <Link href="/manajemen-user" className={cn("inline-flex h-8 items-center whitespace-nowrap rounded-md bg-[#07813e] px-3 text-[11px] font-semibold text-white shadow-sm transition-colors hover:bg-[#066c35]", pathname === "/manajemen-user" && "bg-[#066c35]")}>Kelola User</Link>}
        </div>
      </div>
    </header>
  );
};

export default AppHeader;

"use client";

import { FormEvent, useState } from "react";
import { Link, useRouter } from "@/i18n/navigation";
import {
  CURRENT_USER_STORAGE_KEY,
  USER_STORAGE_KEY,
  type ManagedUser,
} from "@/data/userData";

  const router = useRouter();

  const [username, setUsername] = useState("");
  const [password, setPassword] = useState("");
  const [showPassword, setShowPassword] = useState(false);
  const [rememberMe, setRememberMe] = useState(false);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState("");

  async function handleSubmit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    setError("");

    if (!username.trim() || !password) {
      setError("Username dan password wajib diisi.");
      return;
    }

    setLoading(true);

    try {
      const response = await fetch("/api/auth/login", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          username: username.trim(),
          password,
          rememberMe,
        }),
      });

      const result = await response.json();

      if (!response.ok || !result.success || !result.user) {
        setError(
          result.message || "Username atau password tidak sesuai."
        );
        return;
      }

      const managedUsers: ManagedUser[] = JSON.parse(
        window.localStorage.getItem(USER_STORAGE_KEY) || "[]"
      );

      const managedUser = managedUsers.find(
        (user) =>
          String(user.username).trim().toLowerCase() ===
          String(result.user.username).trim().toLowerCase()
      );

      const currentUser = {
        ...result.user,
        ...(managedUser
          ? {
              name: managedUser.name,
              username: managedUser.username,
              email: managedUser.email,
              gender: managedUser.gender,
            }
          : {}),
      };

      window.localStorage.setItem(
        CURRENT_USER_STORAGE_KEY,
        JSON.stringify(currentUser)
      );

      window.dispatchEvent(new Event("ggf-user-updated"));

      router.push("/");
      router.refresh();
    } catch {
      setError(
        "Tidak dapat terhubung ke server. Silakan coba kembali."
      );
    } finally {
      setLoading(false);
    }
  }

    <main className="ggf-login">

            draggable={false}
          />

          <div className="ggf-login-brand-text">
            <strong>REKAP DATA KECELAKAAN KERJA</strong>
            <span>Sistem Informasi Keselamatan Kerja</span>
          </div>

          </div>

          <form onSubmit={handleSubmit} className="ggf-login-form">
            <div className="ggf-login-field">
              <label htmlFor="username">Username</label>

              <input
                id="username"
                name="username"
                type="text"
                autoComplete="username"
                placeholder="Masukkan username"
                value={username}
                onChange={(event) => setUsername(event.target.value)}
                disabled={loading}
                required
              />
            </div>

            <div className="ggf-login-field">
              <label htmlFor="password">Password</label>

                  autoComplete="current-password"
                  placeholder="Masukkan password"
                  value={password}

                  required
